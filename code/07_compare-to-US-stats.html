read_natality <- function(year, filename) {
  suppressWarnings({
    readxl::read_excel(filename, sheet = as.character(year), ) %>%
      janitor::clean_names() %>%
      select(-notes, -ends_with("code"), -any_of("percent_of_total_births")) %>%
      filter(!is.na(state_of_residence)) %>%
      mutate(year = year)
  })
}
